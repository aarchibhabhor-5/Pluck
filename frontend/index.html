// Pluck snippet: useDebounce — react ^18
import { useState, useEffect } from 'react';

export function useDebounce<T>(
  value: T,
  delay: number = 300
): T {
  const [debouncedVal, setVal] =
    useState<T>(value);

  useEffect(() => {
    const t = setTimeout(()
      => setVal(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);

  return debouncedVal;
}
