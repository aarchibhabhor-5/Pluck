Debounces rapidly changing values with a configurable timeout window in React 18 using useEffect cleanup.

CUSTOMIZE VARIABLES
ms
Install dependencies: npm i react@^18.3.1 
import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delay: number = 300): T {
  const [debouncedVal, setVal] = useState<T>(value);

  useEffect(() => {
    const t = setTimeout(() => setVal(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);

  return debouncedVal;
}
